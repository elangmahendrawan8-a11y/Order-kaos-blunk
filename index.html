const sizeOptions = ["S", "M", "L", "XL", "XXL", "3XL", "4XL", "5XL"];

function generateTableRows(dataList, tableId) {
const tbody = document.getElementById(tableId);
tbody.innerHTML = "";

dataList.forEach((item, index) => {
const tr = document.createElement("tr");

// No
const tdNo = document.createElement("td");
tdNo.textContent = index + 1;
tdNo.style.textAlign = "center";
tr.appendChild(tdNo);

// Nama Pemesan
const tdNama = document.createElement("td");
tdNama.textContent = item.nama;
tr.appendChild(tdNama);

// Ukuran Baju
const tdSize = document.createElement("td");
const selectSize = document.createElement("select");
sizeOptions.forEach(opt => {
const option = document.createElement("option");
option.value = opt;
option.textContent = opt;
if (opt.toUpperCase() === item.size.toUpperCase()) {
option.selected = true;
}
selectSize.appendChild(option);
});
tdSize.appendChild(selectSize);
tr.appendChild(tdSize);

// Status Pembayaran
const tdStatus = document.createElement("td");
const selectStatus = document.createElement("select");
const statusOptions = ["Belum Lunas", "Masih DP", "Lunas"];
statusOptions.forEach(st => {
const option = document.createElement("option");
option.value = st;
option.textContent = st;
if (st === item.status) option.selected = true;
selectStatus.appendChild(option);
});

function updateStatusColor() {
selectStatus.className = "";
if (selectStatus.value === "Lunas") selectStatus.classList.add("status-lunas");
else if (selectStatus.value === "Masih DP") selectStatus.classList.add("status-dp");
else selectStatus.classList.add("status-belum");
}
selectStatus.addEventListener("change", updateStatusColor);
updateStatusColor();

tdStatus.appendChild(selectStatus);
tr.appendChild(tdStatus);

// Keterangan
const tdKet = document.createElement("td");
const inputKet = document.createElement("input");
inputKet.type = "text";
inputKet.value = item.ket || "";
inputKet.placeholder = "Keterangan...";
tdKet.appendChild(inputKet);
tr.appendChild(tdKet);

// Tombol Hapus
const tdAksi = document.createElement("td");
tdAksi.style.textAlign = "center";
const btnDelete = document.createElement("button");
btnDelete.className = "btn-delete";
btnDelete.textContent = "Hapus";
btnDelete.onclick = () => {
dataList.splice(index, 1);
renderAll();
};
tdAksi.appendChild(btnDelete);
tr.appendChild(tdAksi);

tbody.appendChild(tr);
});
}

function renderAll() {
generateTableRows(dataAcara, "table-acara");

}

// Fitur Tambah Pemesan Baru
function tambahPemesan() {
const nama = document.getElementById("new-nama").value.trim();
const kategori = document.getElementById("new-kategori").value;
const size = document.getElementById("new-size").value;
const status = document.getElementById("new-status").value;
const ket = document.getElementById("new-ket").value.trim();

if (!nama) {
alert("Harap isi nama pemesan terlebih dahulu!");
return;
}

const newItem = { nama, size, status, ket };

if (kategori === "acara") {
dataAcara.push(newItem);
} else {
dataKaos.push(newItem);
}

// Reset Form Input
document.getElementById("new-nama").value = "";
document.getElementById("new-ket").value = "";

// Render Ulang Tabel
renderAll();
}
